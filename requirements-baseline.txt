# Baseline-only dependencies — vLLM inference on WebShop test split.
# Use this instead of requirements.txt when only running few-shot eval
# (no SFT/PRM/PPO training).
#
# Why a separate file:
#   - requirements.txt pins `vllm>=0.6` (Llama-3 era) which on a torch>=2.6
#     image will try to downgrade torch and frequently exhausts AutoDL's
#     ~30 GB system disk mid-install.
#   - flash-attn / deepspeed / bitsandbytes / wandb are training-only and
#     add 15+ minutes of build time (flash-attn) for no inference benefit.
#
# Install:
#   pip install --no-cache-dir -r requirements-baseline.txt
#
# Pre-reqs the script needs but pip can't install:
#   - Java (for WebShop's pyserini index) — bash scripts/setup_data.sh
#   - WebShop env editable install — bash scripts/setup_data.sh
#   - en_core_web_lg spacy model — bash scripts/setup_data.sh

# Inference / serving — no upper pin so it picks a wheel matching the
# image's installed torch version (e.g. torch 2.8 ⇒ vllm 0.10+).
vllm
openai>=1.30

# Tokenizer + chat template
transformers>=4.45

# Model download from ModelScope (faster than HF in CN regions)
modelscope

# WebShop env runtime
beautifulsoup4
flask
gym>=0.21
spacy
# pyserini 0.24+ ships Lucene 10 which dropped Lucene 8 index support.
# WebShop's indexes.zip is built with Lucene 8.x, so we need pyserini that
# bundles Lucene 9.x (which still reads 8.x indexes). 0.22.1 is the last
# such version that still has good Python 3.12 wheels.
pyserini==0.22.1

# Utilities
aiohttp
tqdm
numpy
pandas
gdown
